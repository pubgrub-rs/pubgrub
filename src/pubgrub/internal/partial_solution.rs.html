22/// The number of decisions that have been made at some point in solving.
23///
24/// The logical level is zero-based, but it is stored as `level + 1` in a
25/// `NonZeroU32` so that `Option<DecisionLevel>` fits in the same 4 bytes as a
26/// bare `u32`, with `None` occupying the all-zero niche.
27#[derive(Copy, Clone, Ord, PartialOrd, Eq, PartialEq)]
28pub(crate) struct DecisionLevel(NonZeroU32);
29
30impl DecisionLevel {
31    /// Decision level zero: no decisions have been made yet.
32    pub(crate) const ZERO: DecisionLevel = DecisionLevel(NonZeroU32::MIN);
33    /// Sentinel greater than any decision level the solver can reach.
34    pub(crate) const MAX: DecisionLevel = DecisionLevel(NonZeroU32::MAX);
35
36    /// Build a `DecisionLevel` from its zero-based logical value.
37    pub(crate) fn new(level: u32) -> Self {
38        // `level + 1` is always >= 1; the only overflow would be `u32::MAX`
39        // decisions, which the solver cannot reach.
40        Self(NonZeroU32::new(level + 1).expect("decision level overflow"))
41    }
42
43    /// The zero-based logical decision level.
44    pub(crate) fn get(self) -> u32 {
45        self.0.get() - 1
46    }
47
48    /// Advances to the next decision level, panicking if the representation is exhausted.
49    pub(crate) fn increment(self) -> Self {
50        Self(self.0.checked_add(1).expect("decision level overflow"))
51    }
52}
53
54impl Debug for DecisionLevel {
55    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
56        write!(f, "DecisionLevel({})", self.get())
57    }
58}

61const _: () = assert!(std::mem::size_of::<Option<DecisionLevel>>() == std::mem::size_of::<u32>());

63/// The partial solution contains all package assignments,
64/// organized by package and historically ordered.
65#[derive(Clone, Debug)]
66pub(crate) struct PartialSolution<DP: DependencyProvider> {
67    next_global_index: u32,
68    /// The number of decisions that have been made, equal to the number of packages with decisions.
69    current_decision_level: DecisionLevel,
70    /// Store for all known package decisions and package derivations.
71    ///
72    /// "assignment" refers to both packages with decisions and package with only derivations and
73    /// no decision yet. We combine this in a single index map with a decisions section in front and
74    /// a derivations sections in the back, where making a decision moves a package from the
75    /// derivations sections to the decisions section.
76    ///
77    /// `[..current_decision_level]`: Packages that have had a decision made, sorted by the
78    /// `decision_level`. The section is can be seen as the partial solution, it contains a
79    /// mapping from package name to decided version. The sorting makes it very efficient to
80    /// extract the solution, and to backtrack to a particular decision level. The
81    /// `AssignmentsIntersection` is always a `Decision`.
82    ///
83    /// `[current_decision_level..]`: Packages that are dependencies of some other package,
84    /// but have not yet been decided. The `AssignmentsIntersection` is always a `Derivations`, the
85    /// derivations store the obligations from the decided packages.
86    #[allow(clippy::type_complexity)]
87    package_assignments: FnvIndexMap<Id<DP::P>, PackageAssignments<DP::P, DP::VS, DP::M>>,
88    /// The undecided packages order by their `Priority`.
89    ///
90    /// The max heap allows quickly `pop`ing the highest priority package.
91    ///
92    /// The `Reverse<u32>` is the discovery order of packages used as tiebreaker. Its order is that
93    /// of a breadth-first search.
94    #[allow(clippy::type_complexity)]
95    prioritized_potential_packages:
96        PriorityQueue<Id<DP::P>, (DP::Priority, Reverse<u32>), BuildHasherDefault<FxHasher>>,
97    /// Packages whose derivations changed since the last time `prioritize` was called and need
98    /// their priorities to be updated.
99    outdated_priorities: FnvIndexSet<Id<DP::P>>,
100    /// For each completed backtrack generation, the highest decision level that remains valid.
101    /// The active generation is represented by the missing entry after the end of this vector.
102    last_valid_decision_levels: Vec<DecisionLevel>,
103}

105/// A package assignment is either a decision or a list of (accumulated) derivations without a
106/// decision.
107#[derive(Clone, Debug)]
108struct PackageAssignments<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> {
109    /// Whether the assigment is a decision or a derivation.
110    assignments_intersection: AssignmentsIntersection<VS>,
111    /// All constraints on the package version from previous decisions, accumulated by decision
112    /// level.
113    dated_derivations: SmallVec<DatedDerivation<P, VS, M>>,
114    /// Smallest [`DecisionLevel`] in `dated_derivations`.
115    smallest_decision_level: DecisionLevel,
116    /// Highest [`DecisionLevel`] in `dated_derivations`.
117    highest_decision_level: DecisionLevel,
118}
119
120impl<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> Display
121    for PackageAssignments<P, VS, M>
122{
123    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
124        let derivations: Vec<_> = self
125            .dated_derivations
126            .iter()
127            .map(|dd| dd.to_string())
128            .collect();
129        write!(
130            f,
131            "decision range: {:?}..{:?}\nderivations:\n  {}\n,assignments_intersection: {}",
132            self.smallest_decision_level,
133            self.highest_decision_level,
134            derivations.join("\n  "),
135            self.assignments_intersection
136        )
137    }
138}

140#[derive(Clone, Debug)]
141struct DatedDerivation<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> {
142    global_index: u32,
143    /// Only decisions up this level has been used to compute the accumulated term.
144    decision_level: DecisionLevel,
145    cause: IncompId<P, VS, M>,
146    /// The intersection of all terms up to `decision_level`.
147    ///
148    /// It may not contain all terms of this `decision_level`, there may be more than one
149    /// `DatedDerivation` per decision level.
150    accumulated_intersection: Term<VS>,
151}
152
153impl<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> Display
154    for DatedDerivation<P, VS, M>
155{
156    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
157        write!(f, "{:?}, cause: {:?}", self.decision_level, self.cause)
158    }
159}

161#[derive(Clone, Debug)]
162enum AssignmentsIntersection<VS: VersionSet> {
163    /// A decision on package for version has been made at the given level.
164    Decision {
165        decision_level: u32,
166        version: VS::V,
167        /// The version, but as positive singleton term.
168        term: Term<VS>,
169    },
170    Derivations(Term<VS>),
171}
172
173impl<VS: VersionSet> Display for AssignmentsIntersection<VS> {
174    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
175        match self {
176            Self::Decision {
177                decision_level,
178                version,
179                term: _,
180            } => {
181                write!(f, "Decision: level {decision_level}, v = {version}")
182            }
183            Self::Derivations(term) => write!(f, "Derivations term: {term}"),
184        }
185    }
186}

260    /// Add a decision.
261    pub(crate) fn add_decision(&mut self, package: Id<DP::P>, version: DP::V) {
262        // Check that add_decision is never used in the wrong context.
263        if cfg!(debug_assertions) {
264            match self.package_assignments.get_mut(&package) {
265                None => panic!("Derivations must already exist"),
266                Some(pa) => match &pa.assignments_intersection {
267                    // Cannot be called when a decision has already been taken.
268                    AssignmentsIntersection::Decision { .. } => {
269                        panic!("Already existing decision")
270                    }
271                    // Cannot be called if the versions is not contained in the terms' intersection.
272                    AssignmentsIntersection::Derivations(term) => {
273                        debug_assert!(
274                            term.contains(&version),
275                            "{package:?}: {version} was expected to be contained in {term}",
276                        )
277                    }
278                },
279            }
280        }
281        let new_idx = self.current_decision_level.get() as usize;
282        self.current_decision_level = self.current_decision_level.increment();
283        let (old_idx, _, pa) = self
284            .package_assignments
285            .get_full_mut(&package)
286            .expect("Derivations must already exist");
287        pa.highest_decision_level = self.current_decision_level;
288        pa.assignments_intersection = AssignmentsIntersection::Decision {
289            decision_level: self.next_global_index,
290            version: version.clone(),
291            term: Term::exact(version),
292        };
293        // Maintain that the beginning of the `package_assignments` Have all decisions in sorted order.
294        if new_idx != old_idx {
295            self.package_assignments.swap_indices(new_idx, old_idx);
296        }
297        self.next_global_index += 1;
298    }

300    /// Add a derivation.
301    pub(crate) fn add_derivation(
302        &mut self,
303        package: Id<DP::P>,
304        cause: IncompDpId<DP>,
305        store: &Arena<Incompatibility<DP::P, DP::VS, DP::M>>,
306    ) {
307        use indexmap::map::Entry;
308        let mut dated_derivation = DatedDerivation {
309            global_index: self.next_global_index,
310            decision_level: self.current_decision_level,
311            cause,
312            accumulated_intersection: store[cause].get(package).unwrap().negate(),
313        };
314        self.next_global_index += 1;
315        match self.package_assignments.entry(package) {
316            Entry::Occupied(mut occupied) => {
317                let pa = occupied.get_mut();
318                pa.highest_decision_level = self.current_decision_level;
319                match &mut pa.assignments_intersection {
320                    // Check that add_derivation is never called in the wrong context.
321                    AssignmentsIntersection::Decision { .. } => {
322                        panic!("add_derivation should not be called after a decision")
323                    }
324                    AssignmentsIntersection::Derivations(t) => {
325                        *t = t.intersection(&dated_derivation.accumulated_intersection);
326                        dated_derivation.accumulated_intersection = t.clone();
327                        if t.is_positive() {
328                            self.outdated_priorities.insert(package);
329                        }
330                    }
331                }
332                pa.dated_derivations.push(dated_derivation);
333            }
334            Entry::Vacant(v) => {
335                let term = dated_derivation.accumulated_intersection.clone();
336                if term.is_positive() {
337                    self.outdated_priorities.insert(package);
338                }
339                v.insert(PackageAssignments {
340                    smallest_decision_level: self.current_decision_level,
341                    highest_decision_level: self.current_decision_level,
342                    dated_derivations: SmallVec::One([dated_derivation]),
343                    assignments_intersection: AssignmentsIntersection::Derivations(term),
344                });
345            }
346        }
347    }

349    #[cold]
350    pub(crate) fn pick_highest_priority_pkg(
351        &mut self,
352        mut prioritizer: impl FnMut(Id<DP::P>, &DP::VS) -> DP::Priority,
353    ) -> Option<(Id<DP::P>, &DP::VS)> {
354        let prioritized_potential_packages = &mut self.prioritized_potential_packages;
355        while let Some(p) = self.outdated_priorities.pop() {
356            let Some(pa) = self.package_assignments.get(&p) else {
357                continue;
358            };
359            let Some(r) = pa.assignments_intersection.potential_package_filter() else {
360                continue;
361            };
362            let priority = prioritizer(p, r);
363            prioritized_potential_packages.push(p, (priority, Reverse(p.into_raw() as u32)));
364        }
365        while let Some(p) = self.prioritized_potential_packages.pop().map(|(p, _)| p) {
366            let Some(pa) = self.package_assignments.get(&p) else {
367                continue;
368            };
369            if let Some(r) = pa.assignments_intersection.potential_package_filter() {
370                return Some((p, r));
371            }
372        }
373        None
374    }

376    /// If a partial solution has, for every positive derivation,
377    /// a corresponding decision that satisfies that assignment,
378    /// it's a total solution and version solving has succeeded.
379    pub(crate) fn extract_solution(&self) -> impl Iterator<Item = (Id<DP::P>, DP::V)> + '_ {
380        self.package_assignments
381            .iter()
382            .take(self.current_decision_level.get() as usize)
383            .map(|(&p, pa)| match &pa.assignments_intersection {
384                AssignmentsIntersection::Decision {
385                    decision_level: _,
386                    version: v,
387                    term: _,
388                } => (p, v.clone()),
389                AssignmentsIntersection::Derivations(_) => {
390                    // The invariant on the order in `self.package_assignments` was broken.
391                    let mut context = String::new();
392                    for (id, assignment) in self
393                        .package_assignments
394                        .iter()
395                        .take(self.current_decision_level.get() as usize)
396                    {
397                        context.push_str(&format!(
398                            " * {:?} {:?}\n",
399                            id, assignment.assignments_intersection
400                        ));
401                    }
402                    panic!(
403                        "Derivations in the Decision part. Decision level {}\n{}",
404                        self.current_decision_level.get(),
405                        context
406                    )
407                }
408            })
409    }

411    /// Backtrack the partial solution to a given decision level.
412    pub(crate) fn backtrack(&mut self, decision_level: DecisionLevel) {
413        self.current_decision_level = decision_level;
414        self.package_assignments.retain(|p, pa| {
415            if pa.smallest_decision_level > decision_level {
416                // Remove all entries that have a smallest decision level higher than the backtrack target.
417                false
418            } else if pa.highest_decision_level <= decision_level {
419                // Do not change entries older than the backtrack decision level target.
420                if pa
421                    .assignments_intersection
422                    .potential_package_filter()
423                    .is_some()
424                    && self.prioritized_potential_packages.get(p).is_none()
425                {
426                    self.outdated_priorities.insert(*p);
427                }
428                true
429            } else {
430                // smallest_decision_level <= decision_level < highest_decision_level
431                //
432                // Since decision_level < highest_decision_level,
433                // We can be certain that there will be no decision in this package assignments
434                // after backtracking, because such decision would have been the last
435                // assignment and it would have the "highest_decision_level".
436
437                // Truncate the history.
438                while pa.dated_derivations.last().map(|dd| dd.decision_level) > Some(decision_level)
439                {
440                    pa.dated_derivations.pop();
441                }
442                debug_assert!(!pa.dated_derivations.is_empty());
443
444                let last = pa.dated_derivations.last().unwrap();
445
446                // Update highest_decision_level.
447                pa.highest_decision_level = last.decision_level;
448
449                // Reset the assignments intersection.
450                pa.assignments_intersection =
451                    AssignmentsIntersection::Derivations(last.accumulated_intersection.clone());
452
453                self.prioritized_potential_packages.remove(p);
454                if pa.assignments_intersection.term().is_positive() {
455                    self.outdated_priorities.insert(*p);
456                }
457                true
458            }
459        });
460
461        // Close the active generation, then lower every generation invalidated by this backtrack
462        // to the new highest valid decision level.
463        self.last_valid_decision_levels.push(DecisionLevel::MAX);
464        let index = self
465            .last_valid_decision_levels
466            .partition_point(|&level| level <= decision_level);
467        self.last_valid_decision_levels[index..].fill(decision_level);
468    }

470    /// Add a package version as decision if none of its dependencies conflicts with the partial
471    /// solution.
472    ///
473    /// If the resolution never backtracked before, a fast path adds the package version directly
474    /// without checking dependencies.
475    ///
476    /// Returns the incompatibility that caused the current version to be rejected, if a conflict
477    /// in the dependencies was found.
478    pub(crate) fn add_package_version_incompatibilities(
479        &mut self,
480        package: Id<DP::P>,
481        version: DP::V,
482        new_incompatibilities: std::ops::Range<IncompId<DP::P, DP::VS, DP::M>>,
483        store: &Arena<Incompatibility<DP::P, DP::VS, DP::M>>,
484    ) -> Option<IncompId<DP::P, DP::VS, DP::M>> {
485        if !self.has_backtracked() {
486            // Fast path: Nothing has yet gone wrong during this resolution. This call is unlikely to be the first problem.
487            // So let's live with a little bit of risk and add the decision without checking the dependencies.
488            // The worst that can happen is we will have to do a full backtrack which only removes this one decision.
489            log::info!("add_decision: {package:?} @ {version} without checking dependencies");
490            self.add_decision(package, version);
491            return None;
492        }
493
494        // Check if any of the dependencies preclude deciding on this crate version.
495        let package_term = Term::exact(version.clone());
496        let relation = |incompat: IncompId<DP::P, DP::VS, DP::M>| {
497            store[incompat].relation(|p| {
498                // The current package isn't part of the package assignments yet.
499                if p == package {
500                    Some(&package_term)
501                } else {
502                    self.term_intersection_for_package(p)
503                }
504            })
505        };
506        if let Some(satisfied) = Id::range_to_iter(new_incompatibilities)
507            .find(|incompat| relation(*incompat) == Relation::Satisfied)
508        {
509            log::info!(
510                "rejecting decision {package:?} @ {version} because its dependencies conflict"
511            );
512            Some(satisfied)
513        } else {
514            log::info!("adding decision: {package:?} @ {version}");
515            self.add_decision(package, version);
516            None
517        }
518    }

525        incompat.relation(|package| self.term_intersection_for_package(package))

538    /// Figure out if the satisfier and previous satisfier are of different decision levels.
539    #[allow(clippy::type_complexity)]
540    pub(crate) fn satisfier_search(
541        &self,
542        incompat: &Incompatibility<DP::P, DP::VS, DP::M>,
543        store: &Arena<Incompatibility<DP::P, DP::VS, DP::M>>,
544    ) -> (Id<DP::P>, SatisfierSearch<DP::P, DP::VS, DP::M>) {
545        let satisfied_map = Self::find_satisfier(incompat, &self.package_assignments);
546        let (&satisfier_package, &(satisfier_cause, _, satisfier_decision_level)) = satisfied_map
547            .iter()
548            .max_by_key(|(_p, (_, global_index, _))| global_index)
549            .unwrap();
550        let previous_satisfier_level = Self::find_previous_satisfier(
551            incompat,
552            satisfier_package,
553            satisfied_map,
554            &self.package_assignments,
555            store,
556        );
557        let search_result = if previous_satisfier_level >= satisfier_decision_level {
558            SatisfierSearch::SameDecisionLevels {
559                satisfier_cause: satisfier_cause.unwrap(),
560            }
561        } else {
562            SatisfierSearch::DifferentDecisionLevels {
563                previous_satisfier_level,
564            }
565        };
566        (satisfier_package, search_result)
567    }

569    /// A satisfier is the earliest assignment in partial solution such that the incompatibility
570    /// is satisfied by the partial solution up to and including that assignment.
571    ///
572    /// Returns a map indicating for each package term, when that was first satisfied in history.
573    /// If we effectively found a satisfier, the returned map must be the same size that incompat.
574    ///
575    /// Question: This is possible since we added a "global_index" to every dated_derivation.
576    /// It would be nice if we could get rid of it, but I don't know if then it will be possible
577    /// to return a coherent previous_satisfier_level.
578    #[allow(clippy::type_complexity)]
579    fn find_satisfier(
580        incompat: &Incompatibility<DP::P, DP::VS, DP::M>,
581        package_assignments: &FnvIndexMap<Id<DP::P>, PackageAssignments<DP::P, DP::VS, DP::M>>,
582    ) -> SatisfiedMap<DP::P, DP::VS, DP::M> {
583        let mut satisfied = SmallMap::Empty;
584        for (package, incompat_term) in incompat.iter() {
585            let pa = package_assignments.get(&package).expect("Must exist");
586            satisfied.insert(package, pa.satisfier(package, &incompat_term.negate()));
587        }
588        satisfied
589    }

591    /// Earliest assignment in the partial solution before satisfier
592    /// such that incompatibility is satisfied by the partial solution up to
593    /// and including that assignment plus satisfier.
594    #[allow(clippy::type_complexity)]
595    fn find_previous_satisfier(
596        incompat: &Incompatibility<DP::P, DP::VS, DP::M>,
597        satisfier_package: Id<DP::P>,
598        mut satisfied_map: SatisfiedMap<DP::P, DP::VS, DP::M>,
599        package_assignments: &FnvIndexMap<Id<DP::P>, PackageAssignments<DP::P, DP::VS, DP::M>>,
600        store: &Arena<Incompatibility<DP::P, DP::VS, DP::M>>,
601    ) -> DecisionLevel {
602        // First, let's retrieve the previous derivations and the initial accum_term.
603        let satisfier_pa = package_assignments.get(&satisfier_package).unwrap();
604        let (satisfier_cause, _gidx, _dl) = satisfied_map.get(&satisfier_package).unwrap();
605
606        let accum_term = if let &Some(cause) = satisfier_cause {
607            store[cause].get(satisfier_package).unwrap().negate()
608        } else {
609            match &satisfier_pa.assignments_intersection {
610                AssignmentsIntersection::Derivations(_) => panic!("must be a decision"),
611                AssignmentsIntersection::Decision {
612                    decision_level: _,
613                    version: _,
614                    term,
615                } => term.clone(),
616            }
617        };
618
619        let incompat_term = incompat
620            .get(satisfier_package)
621            .expect("satisfier package not in incompat");
622
623        satisfied_map.insert(
624            satisfier_package,
625            satisfier_pa.satisfier(
626                satisfier_package,
627                &accum_term.intersection(&incompat_term.negate()),
628            ),
629        );
630
631        // Finally, let's identify the decision level of that previous satisfier.
632        let (_, &(_, _, decision_level)) = satisfied_map
633            .iter()
634            .max_by_key(|(_p, (_, global_index, _))| global_index)
635            .unwrap();
636        decision_level.max(DecisionLevel::new(1))
637    }
638
639    pub(crate) fn current_decision_level(&self) -> DecisionLevel {
640        self.current_decision_level
641    }
642}
643
644impl<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> PackageAssignments<P, VS, M> {
645    fn satisfier(
646        &self,
647        package: Id<P>,
648        start_term: &Term<VS>,
649    ) -> (Option<IncompId<P, VS, M>>, u32, DecisionLevel) {
650        let empty = Term::empty();
651        // Indicate if we found a satisfier in the list of derivations, otherwise it will be the decision.
652        let idx = self
653            .dated_derivations
654            .as_slice()
655            .partition_point(|dd| !dd.accumulated_intersection.is_disjoint(start_term));
656        if let Some(dd) = self.dated_derivations.get(idx) {
657            debug_assert_eq!(dd.accumulated_intersection.intersection(start_term), empty);
658            return (Some(dd.cause), dd.global_index, dd.decision_level);
659        }
660        // If it wasn't found in the derivations,
661        // it must be the decision which is last (if called in the right context).
662        match &self.assignments_intersection {
663            AssignmentsIntersection::Decision {
664                decision_level: global_index,
665                version: _,
666                term: _,
667            } => (None, *global_index, self.highest_decision_level),
668            AssignmentsIntersection::Derivations(accumulated_intersection) => {
669                unreachable!(
670                    concat!(
671                        "while processing package {:?}: ",
672                        "accum_term = {} has overlap with incompat_term = {}, ",
673                        "which means the last assignment should have been a decision, ",
674                        "but instead it was a derivation. This shouldn't be possible! ",
675                        "(Maybe your Version ordering is broken?)"
676                    ),
677                    package, accumulated_intersection, start_term
678                )
679            }
680        }
681    }
682}
683
684impl<VS: VersionSet> AssignmentsIntersection<VS> {
685    /// Returns the term intersection of all assignments (decision included).
686    fn term(&self) -> &Term<VS> {
687        match self {
688            Self::Decision {
689                decision_level: _,
690                version: _,
691                term,
692            } => term,
693            Self::Derivations(term) => term,
694        }
695    }
696
697    /// A package is a potential pick if there isn't an already
698    /// selected version (no "decision")
699    /// and if it contains at least one positive derivation term
700    /// in the partial solution.
701    fn potential_package_filter(&self) -> Option<&VS> {
702        match self {
703            Self::Decision { .. } => None,
704            Self::Derivations(term_intersection) => {
705                if term_intersection.is_positive() {
706                    Some(term_intersection.unwrap_positive())
707                } else {
708                    None
709                }
710            }
711        }
712    }
713}
