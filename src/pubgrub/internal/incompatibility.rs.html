15#[derive(Debug, Clone)]
16struct ContradictionCache {
17    /// The decision level where this incompatibility became contradicted.
18    decision_level: DecisionLevel,
19    /// The backtrack generation for that decision level.
20    backtrack_generation: u32,
21}
22
23impl ContradictionCache {
24    fn not_contradicted() -> Self {
25        Self {
26            decision_level: DecisionLevel::MAX,
27            backtrack_generation: 0,
28        }
29    }
30
31    fn is_contradicted(&self, last_valid_decision_levels: &[DecisionLevel]) -> bool {
32        // The active generation has no backtrack target yet, so every contradiction recorded in
33        // it remains valid. Completed generations store the highest decision level that survived
34        // their first invalidating backtrack.
35        last_valid_decision_levels
36            .get(self.backtrack_generation as usize)
37            .map(|&level| self.decision_level <= level)
38            .unwrap_or(true)
39    }
40}

42/// An incompatibility is a set of terms for different packages
43/// that should never be satisfied all together.
44/// An incompatibility usually originates from a package dependency.
45/// For example, if package A at version 1 depends on package B
46/// at version 2, you can never have both terms `A = 1`
47/// and `not B = 2` satisfied at the same time in a partial solution.
48/// This would mean that we found a solution with package A at version 1
49/// but not with package B at version 2.
50/// Yet A at version 1 depends on B at version 2 so this is not possible.
51/// Therefore, the set `{ A = 1, not B = 2 }` is an incompatibility,
52/// defined from dependencies of A at version 1.
53///
54/// Incompatibilities can also be derived from two other incompatibilities
55/// during conflict resolution. More about all this in
56/// [PubGrub documentation](https://github.com/dart-lang/pub/blob/master/doc/solver.md#incompatibility).
57#[derive(Debug, Clone)]
58pub(crate) struct Incompatibility<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> {
59    package_terms: SmallMap<Id<P>, Term<VS>>,
60    kind: Kind<P, VS, M>,
61    contradiction_cache: ContradictionCache,
62}
63
64/// Type alias of unique identifiers for incompatibilities.
65pub(crate) type IncompId<P, VS, M> = Id<Incompatibility<P, VS, M>>;
66
67pub(crate) type IncompDpId<DP> = IncompId<
68    <DP as DependencyProvider>::P,
69    <DP as DependencyProvider>::VS,
70    <DP as DependencyProvider>::M,
71>;

73#[derive(Debug, Clone)]
74enum Kind<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> {
75    /// Initial incompatibility aiming at picking the root package for the first decision.
76    ///
77    /// This incompatibility drives the resolution, it requires that we pick the (virtual) root
78    /// packages.
79    NotRoot(Id<P>, VS::V),
80    /// There are no versions in the given range for this package.
81    ///
82    /// This incompatibility is used when we tried all versions in a range and no version
83    /// worked, so we have to backtrack
84    NoVersions(Id<P>, VS),
85    /// Incompatibility coming from the dependencies of a given package.
86    ///
87    /// If a@1 depends on b>=1,<2, we create an incompatibility with terms `{a 1, b <1,>=2}` with
88    /// kind `FromDependencyOf(a, b)`. The version sets are stored in the incompatibility terms.
89    ///
90    /// We can merge multiple dependents with the same version. For example, if a@1 depends on b and
91    /// a@2 depends on b, we can say instead a@1||2 depends on b.
92    FromDependencyOf(Id<P>, Id<P>),
93    /// Derived from two causes. Stores cause ids.
94    ///
95    /// For example, if a -> b and b -> c, we can derive a -> c.
96    DerivedFrom(IncompId<P, VS, M>, IncompId<P, VS, M>),
97    /// The package is unavailable for reasons outside pubgrub.
98    ///
99    /// Examples:
100    /// * The version would require building the package, but builds are disabled.
101    /// * The package is not available in the cache, but internet access has been disabled.
102    Custom(Id<P>, VS, M),
103}

105/// A Relation describes how a set of terms can be compared to an incompatibility.
106/// Typically, the set of terms comes from the partial solution.
107#[derive(Eq, PartialEq, Debug)]
108pub(crate) enum Relation<P: Package> {
109    /// We say that a set of terms S satisfies an incompatibility I
110    /// if S satisfies every term in I.
111    Satisfied,
112    /// We say that S contradicts I
113    /// if S contradicts at least one term in I.
114    Contradicted(Id<P>),
115    /// If S satisfies all but one of I's terms and is inconclusive for the remaining term,
116    /// we say S "almost satisfies" I and we call the remaining term the "unsatisfied term".
117    AlmostSatisfied(Id<P>),
118    /// Otherwise, we say that their relation is inconclusive.
119    Inconclusive,
120}

122impl<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> Incompatibility<P, VS, M> {
123    /// Create the initial "not Root" incompatibility.
124    pub(crate) fn not_root(package: Id<P>, version: VS::V) -> Self {
125        Self {
126            package_terms: SmallMap::One([(
127                package,
128                Term::Negative(VS::singleton(version.clone())),
129            )]),
130            kind: Kind::NotRoot(package, version),
131            contradiction_cache: ContradictionCache::not_contradicted(),
132        }
133    }
134
135    /// Create an incompatibility to remember that a given set does not contain any version.
136    pub(crate) fn no_versions(package: Id<P>, term: Term<VS>) -> Self {
137        let set = match &term {
138            Term::Positive(r) => r.clone(),
139            Term::Negative(_) => panic!("No version should have a positive term"),
140        };
141        Self {
142            package_terms: SmallMap::One([(package, term)]),
143            kind: Kind::NoVersions(package, set),
144            contradiction_cache: ContradictionCache::not_contradicted(),
145        }
146    }
147
148    /// Create an incompatibility for a reason outside pubgrub.
149    #[allow(dead_code)] // Used by uv
150    pub(crate) fn custom_term(package: Id<P>, term: Term<VS>, metadata: M) -> Self {
151        let set = match &term {
152            Term::Positive(r) => r.clone(),
153            Term::Negative(_) => panic!("No version should have a positive term"),
154        };
155        Self {
156            package_terms: SmallMap::One([(package, term)]),
157            kind: Kind::Custom(package, set, metadata),
158            contradiction_cache: ContradictionCache::not_contradicted(),
159        }
160    }
161
162    /// Create an incompatibility for a reason outside pubgrub.
163    pub(crate) fn custom_version(package: Id<P>, version: VS::V, metadata: M) -> Self {
164        let set = VS::singleton(version);
165        let term = Term::Positive(set.clone());
166        Self {
167            package_terms: SmallMap::One([(package, term)]),
168            kind: Kind::Custom(package, set, metadata),
169            contradiction_cache: ContradictionCache::not_contradicted(),
170        }
171    }
172
173    /// Build an incompatibility from a given dependency.
174    pub(crate) fn from_dependency(package: Id<P>, versions: VS, dep: (Id<P>, VS)) -> Self {
175        let (p2, set2) = dep;
176        Self {
177            package_terms: if set2 == VS::empty() {
178                SmallMap::One([(package, Term::Positive(versions))])
179            } else {
180                SmallMap::Two([
181                    (package, Term::Positive(versions)),
182                    (p2, Term::Negative(set2)),
183                ])
184            },
185            kind: Kind::FromDependencyOf(package, p2),
186            contradiction_cache: ContradictionCache::not_contradicted(),
187        }
188    }
189
190    fn dependency_terms(&self, p1: Id<P>, p2: Id<P>) -> (&VS, Option<&VS>) {
191        let mut terms = self.package_terms.iter();
192        let versions = match terms.next() {
193            Some((term_package, Term::Positive(versions))) if *term_package == p1 => versions,
194            _ => panic!("dependency incompatibility must start with its positive term"),
195        };
196        let dependency_versions = match terms.next() {
197            None => None,
198            Some((term_package, Term::Negative(versions))) if *term_package == p2 => Some(versions),
199            _ => panic!("dependency incompatibility must end with its negative term"),
200        };
201        assert!(
202            terms.next().is_none(),
203            "dependency incompatibility must contain at most two terms"
204        );
205        (versions, dependency_versions)
206    }
207
208    pub(crate) fn as_dependency(&self) -> Option<(Id<P>, Id<P>)> {
209        match &self.kind {
210            Kind::FromDependencyOf(p1, p2) => Some((*p1, *p2)),
211            _ => None,
212        }
213    }
214
215    /// Merge dependant versions with the same dependency.
216    ///
217    /// When multiple versions of a package depend on the same range of another package,
218    /// we can merge the two into a single incompatibility.
219    /// For example, if a@1 depends on b and a@2 depends on b, we can say instead
220    /// a@1||2 depends on b.
221    ///
222    /// It is a special case of prior cause computation where the unified package
223    /// is the common dependant in the two incompatibilities expressing dependencies.
224    pub(crate) fn merge_dependents(&self, other: &Self) -> Option<Self> {
225        // It is almost certainly a bug to call this method without checking that self is a dependency
226        debug_assert!(self.as_dependency().is_some());
227        // Check that both incompatibilities are of the shape p1 depends on p2,
228        // with the same p1 and p2.
229        let self_pkgs = self.as_dependency()?;
230        if self_pkgs != other.as_dependency()? {
231            return None;
232        }
233        let (p1, p2) = self_pkgs;
234        // We ignore self-dependencies. They are always either trivially true or trivially false,
235        // as the package version implies whether the constraint will always be fulfilled or always
236        // violated.
237        // At time of writing, the public crate API only allowed a map of dependencies,
238        // meaning it can't hit this branch, which requires two self-dependencies.
239        if p1 == p2 {
240            return None;
241        }
242        let dep_term = self.get(p2);
243        // The dependency range for p2 must be the same in both case
244        // to be able to merge multiple p1 ranges.
245        if dep_term != other.get(p2) {
246            return None;
247        }
248        Some(Self::from_dependency(
249            p1,
250            self.get(p1)
251                .unwrap()
252                .unwrap_positive()
253                .union(other.get(p1).unwrap().unwrap_positive()), // It is safe to `simplify` here
254            (
255                p2,
256                dep_term.map_or(VS::empty(), |v| v.unwrap_negative().clone()),
257            ),
258        ))
259    }
260
261    /// Prior cause of two incompatibilities using the rule of resolution.
262    pub(crate) fn prior_cause(
263        incompat: Id<Self>,
264        satisfier_cause: Id<Self>,
265        package: Id<P>,
266        incompatibility_store: &Arena<Self>,
267    ) -> Self {
268        let kind = Kind::DerivedFrom(incompat, satisfier_cause);
269        // Optimization to avoid cloning and dropping t1
270        let (t1, mut package_terms) = incompatibility_store[incompat]
271            .package_terms
272            .split_one(&package)
273            .unwrap();
274        let satisfier_cause_terms = &incompatibility_store[satisfier_cause].package_terms;
275        package_terms.merge(
276            satisfier_cause_terms.iter().filter(|(p, _)| p != &&package),
277            |t1, t2| Some(t1.intersection(t2)),
278        );
279        let term = t1.union(satisfier_cause_terms.get(&package).unwrap());
280        if term != Term::any() {
281            package_terms.insert(package, term);
282        }
283        Self {
284            package_terms,
285            kind,
286            contradiction_cache: ContradictionCache::not_contradicted(),
287        }
288    }
289
290    pub(crate) fn is_contradicted(&self, last_valid_decision_levels: &[DecisionLevel]) -> bool {
291        self.contradiction_cache
292            .is_contradicted(last_valid_decision_levels)
293    }

296        &mut self,
297        decision_level: DecisionLevel,
298        backtrack_generation: u32,

300        self.contradiction_cache = ContradictionCache {

302            backtrack_generation,

306    pub(crate) fn reset_contradiction_cache(&mut self) {

310    /// Check if an incompatibility should mark the end of the algorithm
311    /// because it satisfies the root package.
312    pub(crate) fn is_terminal(&self, root_package: Id<P>, root_version: &VS::V) -> bool {
313        if self.package_terms.len() == 0 {
314            true
315        } else if self.package_terms.len() > 1 {
316            false
317        } else {
318            let (package, term) = self.package_terms.iter().next().unwrap();
319            (package == &root_package) && term.contains(root_version)
320        }
321    }
322
323    /// Get the term related to a given package (if it exists).
324    pub(crate) fn get(&self, package: Id<P>) -> Option<&Term<VS>> {
325        self.package_terms.get(&package)
326    }
327
328    /// Iterate over packages.
329    pub(crate) fn iter(&self) -> impl Iterator<Item = (Id<P>, &Term<VS>)> {
330        self.package_terms
331            .iter()
332            .map(|(package, term)| (*package, term))
333    }

345    /// Build a derivation tree for error reporting.
346    pub(crate) fn build_derivation_tree(
347        self_id: Id<Self>,
348        shared_ids: &Set<Id<Self>>,
349        store: &Arena<Self>,
350        package_store: &HashArena<P>,
351        precomputed: &Map<Id<Self>, Arc<DerivationTree<P, VS, M>>>,
352    ) -> DerivationTree<P, VS, M> {
353        match store[self_id].kind.clone() {
354            Kind::DerivedFrom(id1, id2) => {
355                let derived: Derived<P, VS, M> = Derived {
356                    terms: store[self_id]
357                        .package_terms
358                        .iter()
359                        .map(|(&a, b)| (package_store[a].clone(), b.clone()))
360                        .collect(),
361                    shared_id: shared_ids.get(&self_id).map(|id| id.into_raw()),
362                    cause1: precomputed
363                        .get(&id1)
364                        .expect("Non-topological calls building tree")
365                        .clone(),
366                    cause2: precomputed
367                        .get(&id2)
368                        .expect("Non-topological calls building tree")
369                        .clone(),
370                };
371                DerivationTree::Derived(derived)
372            }
373            Kind::NotRoot(package, version) => {
374                DerivationTree::External(External::NotRoot(package_store[package].clone(), version))
375            }
376            Kind::NoVersions(package, set) => DerivationTree::External(External::NoVersions(
377                package_store[package].clone(),
378                set.clone(),
379            )),
380            Kind::FromDependencyOf(package, dep_package) => {
381                let (package_versions, dependency_versions) =
382                    store[self_id].dependency_terms(package, dep_package);
383                DerivationTree::External(External::FromDependencyOf(
384                    package_store[package].clone(),
385                    package_versions.clone(),
386                    package_store[dep_package].clone(),
387                    dependency_versions.cloned().unwrap_or_else(VS::empty),
388                ))
389            }
390            Kind::Custom(package, set, metadata) => DerivationTree::External(External::Custom(
391                package_store[package].clone(),
392                set.clone(),
393                metadata.clone(),
394            )),
395        }
396    }
397}

399impl<'a, P: Package, VS: VersionSet + 'a, M: Eq + Clone + Debug + Display + 'a>
400    Incompatibility<P, VS, M>
401{
402    /// CF definition of Relation enum.
403    pub(crate) fn relation(&self, terms: impl Fn(Id<P>) -> Option<&'a Term<VS>>) -> Relation<P> {
404        let mut relation = Relation::Satisfied;
405        for (&package, incompat_term) in self.package_terms.iter() {
406            match terms(package).map(|term| incompat_term.relation_with(term)) {
407                Some(term::Relation::Satisfied) => {}
408                Some(term::Relation::Contradicted) => {
409                    return Relation::Contradicted(package);
410                }
411                None | Some(term::Relation::Inconclusive) => {
412                    // If a package is not present, the intersection is the same as [Term::any].
413                    // According to the rules of satisfactions, the relation would be inconclusive.
414                    // It could also be satisfied if the incompatibility term was also [Term::any],
415                    // but we systematically remove those from incompatibilities
416                    // so we're safe on that front.
417                    if relation == Relation::Satisfied {
418                        relation = Relation::AlmostSatisfied(package);
419                    } else {
420                        return Relation::Inconclusive;
421                    }
422                }
423            }
424        }
425        relation
426    }
427}

429impl<P: Package, VS: VersionSet, M: Eq + Clone + Debug + Display> Incompatibility<P, VS, M> {
430    pub fn display<'a>(&'a self, package_store: &'a HashArena<P>) -> impl Display + 'a {
431        match self.iter().collect::<Vec<_>>().as_slice() {
432            [] => "version solving failed".into(),
433            // TODO: special case when that unique package is root.
434            [(package, Term::Positive(range))] => {
435                format!("{} {} is forbidden", package_store[*package], range)
436            }
437            [(package, Term::Negative(range))] => {
438                format!("{} {} is mandatory", package_store[*package], range)
439            }
440            [
441                (p_pos, Term::Positive(r_pos)),
442                (p_neg, Term::Negative(r_neg)),
443            ]
444            | [
445                (p_neg, Term::Negative(r_neg)),
446                (p_pos, Term::Positive(r_pos)),
447            ] => External::<_, _, M>::FromDependencyOf(
448                &package_store[*p_pos],
449                r_pos.clone(),
450                &package_store[*p_neg],
451                r_neg.clone(),
452            )
453            .to_string(),
454            slice => {
455                let str_terms: Vec<_> = slice
456                    .iter()
457                    .map(|(p, t)| format!("{} {}", package_store[*p], t))
458                    .collect();
459                str_terms.join(", ") + " are incompatible"
460            }
461        }
462    }

480    fn contradiction_cache_tracks_backtrack_generations() {
481        let current_generation = ContradictionCache {
482            decision_level: DecisionLevel::new(3),
483            backtrack_generation: 1,
484        };
485
486        // A generation without a recorded backtrack target is still active.
487        assert!(current_generation.is_contradicted(&[DecisionLevel::ZERO]));
488        // Backtracking below the contradiction's decision level invalidates it.
489        assert!(!current_generation.is_contradicted(&[DecisionLevel::ZERO, DecisionLevel::new(2)]));
490        // Backtracking to or above that decision level preserves it.
491        assert!(current_generation.is_contradicted(&[DecisionLevel::ZERO, DecisionLevel::new(3)]));
492
493        let later_generation = ContradictionCache {
494            decision_level: DecisionLevel::new(5),
495            backtrack_generation: 2,
496        };
497        assert!(later_generation.is_contradicted(&[DecisionLevel::ZERO, DecisionLevel::new(3)]));
498        assert!(!later_generation.is_contradicted(&[

500            DecisionLevel::new(3),
501            DecisionLevel::new(4),

699    /// Check that multiple self-dependencies are supported.
700    ///
701    /// The current public API deduplicates dependencies through a map, so we test them here
702    /// manually.
703    ///
704    /// https://github.com/astral-sh/uv/issues/13344
705    #[test]
706    fn package_depend_on_self() {
707        let cases: &[Vec<(String, Ranges<usize>)>] = &[
708            vec![("foo".to_string(), Ranges::full())],
709            vec![
710                ("foo".to_string(), Ranges::full()),
711                ("foo".to_string(), Ranges::full()),
712            ],
713            vec![
714                ("foo".to_string(), Ranges::full()),
715                ("foo".to_string(), Ranges::singleton(1usize)),
716            ],
717            vec![
718                ("foo".to_string(), Ranges::singleton(1usize)),
719                ("foo".to_string(), Ranges::from_range_bounds(1usize..2)),
720                ("foo".to_string(), Ranges::from_range_bounds(1usize..3)),
721            ],
722        ];
723
724        for case in cases {
725            let mut state: State<OfflineDependencyProvider<String, Ranges<usize>>> =
726                State::init("root".to_string(), 0);
727            state.unit_propagation(state.root_package).unwrap();
728
729            // Add the root package
730            state.add_package_version_dependencies(
731                state.root_package,
732                0,
733                Ranges::singleton(0usize),
734                [("foo".to_string(), Ranges::singleton(1usize))],
735            );
736            state.unit_propagation(state.root_package).unwrap();
737
738            // Add a package that depends on itself twice
739            let (next, _) = state
740                .partial_solution
741                .pick_highest_priority_pkg(|_p, _r| (0, Reverse(0)))
742                .unwrap();
743            state.add_package_version_dependencies(
744                next,
745                1,
746                Ranges::singleton(1usize),
747                case.clone(),
748            );
749            state.unit_propagation(next).unwrap();
750
751            assert!(
752                state
753                    .partial_solution
754                    .pick_highest_priority_pkg(|_p, _r| (0, Reverse(0)))
755                    .is_none()
756            );
757
758            let solution: BTreeMap<String, usize> = state
759                .partial_solution
760                .extract_solution()
761                .map(|(p, v)| (state.package_store[p].clone(), v))
762                .collect();
763            let expected = BTreeMap::from([("root".to_string(), 0), ("foo".to_string(), 1)]);
764
765            assert_eq!(solution, expected, "{case:?}");
766        }
767    }
