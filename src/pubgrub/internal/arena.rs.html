9/// The index of a value allocated in an arena that holds `T`s.
10///
11/// The Clone, Copy and other traits are defined manually because
12/// deriving them adds some additional constraints on the `T` generic type
13/// that we actually don't need since it is phantom.
14///
15/// <https://github.com/rust-lang/rust/issues/26925>
16///
17/// The zero-based index is stored as `index + 1` in a [`NonZeroU32`], allowing
18/// `Option<Id<T>>` to use the zero niche and remain four bytes. The largest
19/// representable index is therefore `u32::MAX - 1`.
20pub(crate) struct Id<T> {
21    raw: NonZeroU32,
22    _ty: PhantomData<fn() -> T>,
23}
24
25// Store the zero-based index plus one so `None` can use the zero niche.
26const _: () = assert!(std::mem::size_of::<Option<Id<()>>>() == std::mem::size_of::<u32>());
27
28impl<T> Clone for Id<T> {
29    fn clone(&self) -> Self {
30        *self
31    }
32}
33
34impl<T> Copy for Id<T> {}
35
36impl<T> PartialEq for Id<T> {
37    fn eq(&self, other: &Id<T>) -> bool {
38        self.raw == other.raw
39    }
40}
41
42impl<T> Eq for Id<T> {}
43
44impl<T> Hash for Id<T> {
45    fn hash<H: Hasher>(&self, state: &mut H) {
46        (self.into_raw() as u32).hash(state)
47    }
48}
49
50impl<T> fmt::Debug for Id<T> {
51    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
52        let mut type_name = std::any::type_name::<T>();
53        if let Some(id) = type_name.rfind(':') {
54            type_name = &type_name[id + 1..]
55        }
56        write!(f, "Id::<{}>({})", type_name, self.into_raw())
57    }
58}
59
60impl<T> Id<T> {
61    pub(crate) fn into_raw(self) -> usize {
62        (self.raw.get() - 1) as usize
63    }
64
65    fn from_usize(index: usize) -> Self {
66        assert!(index < u32::MAX as usize, "id index exceeds u32::MAX - 1");
67        Self {
68            raw: NonZeroU32::new(index as u32 + 1).expect("checked id index must be non-zero"),
69            _ty: PhantomData,
70        }
71    }
72
73    pub(crate) fn range_to_iter(range: Range<Self>) -> impl Iterator<Item = Self> {
74        let start = range.start.into_raw();
75        let end = range.end.into_raw();
76        (start..end).map(Self::from_usize)
77    }
78}

80/// Yet another index-based arena.
81///
82/// An arena is a kind of simple grow-only allocator, backed by a `Vec`
83/// where all items have the same lifetime, making it easier
84/// to have references between those items.
85/// They are all dropped at once when the arena is dropped.
86#[derive(Clone, PartialEq, Eq)]
87pub(crate) struct Arena<T> {
88    data: Vec<T>,
89}
90
91impl<T: fmt::Debug> fmt::Debug for Arena<T> {
92    fn fmt(&self, fmt: &mut fmt::Formatter) -> fmt::Result {
93        fmt.debug_struct("Arena")
94            .field("len", &self.data.len())
95            .field("data", &self.data)
96            .finish()
97    }
98}
99
100impl<T> Default for Arena<T> {
101    fn default() -> Self {
102        Self::new()
103    }
104}
105
106impl<T> Arena<T> {
107    pub(crate) fn new() -> Self {
108        Self { data: Vec::new() }
109    }
110
111    pub(crate) fn alloc(&mut self, value: T) -> Id<T> {
112        let id = Id::from_usize(self.data.len());
113        self.data.push(value);
114        id
115    }
116
117    pub(crate) fn alloc_iter<I: Iterator<Item = T>>(&mut self, values: I) -> Range<Id<T>> {
118        let start = Id::from_usize(self.data.len());
119        self.data.reserve(values.size_hint().0);
120        values.for_each(|v| {
121            self.alloc(v);
122        });
123        let end = Id::from_usize(self.data.len());
124        Range { start, end }
125    }
126}

128impl<T> Index<Id<T>> for Arena<T> {
129    type Output = T;
130    fn index(&self, id: Id<T>) -> &T {
131        &self.data[id.into_raw()]
132    }
133}
134
135impl<T> IndexMut<Id<T>> for Arena<T> {
136    fn index_mut(&mut self, id: Id<T>) -> &mut Self::Output {
137        &mut self.data[id.into_raw()]
138    }
139}
140
141impl<T> Index<Range<Id<T>>> for Arena<T> {
142    type Output = [T];
143    fn index(&self, id: Range<Id<T>>) -> &[T] {
144        &self.data[id.start.into_raw()..id.end.into_raw()]
145    }
146}

148/// Yet another index-based arena. This one de-duplicates entries by hashing.
149///
150/// An arena is a kind of simple grow-only allocator, backed by a `Vec`
151/// where all items have the same lifetime, making it easier
152/// to have references between those items.
153/// In this case the `Vec` is inside a `IndexSet` allowing fast lookup by value not just index.
154/// They are all dropped at once when the arena is dropped.
155#[derive(Clone, PartialEq, Eq)]
156pub struct HashArena<T: Hash + Eq> {
157    data: FnvIndexSet<T>,
158}
159
160impl<T: Hash + Eq + fmt::Debug> fmt::Debug for HashArena<T> {
161    fn fmt(&self, fmt: &mut fmt::Formatter) -> fmt::Result {
162        fmt.debug_struct("Arena")
163            .field("len", &self.data.len())
164            .field("data", &self.data)
165            .finish()
166    }
167}
168
169impl<T: Hash + Eq> HashArena<T> {
170    pub fn new() -> Self {
171        Self::default()
172    }
173
174    pub fn alloc(&mut self, value: T) -> Id<T> {
175        assert!(
176            self.data.len() < u32::MAX as usize || self.data.contains(&value),
177            "id index exceeds u32::MAX - 1"
178        );
179        let (raw, _) = self.data.insert_full(value);
180        Id::from_usize(raw)
181    }
182}
183
184impl<T: Hash + Eq> Default for HashArena<T> {
185    fn default() -> Self {
186        Self {
187            data: FnvIndexSet::default(),
188        }
189    }
190}
191
192impl<T: Hash + Eq> Index<Id<T>> for HashArena<T> {
193    type Output = T;
194    fn index(&self, id: Id<T>) -> &T {
195        &self.data[id.into_raw()]
196    }
197}

199#[cfg(test)]
200mod tests {
201    use super::{Arena, Id};
202
203    #[test]
204    fn alloc_iter_preserves_order_and_duplicates() {
205        let mut arena = Arena::new();
206        let existing = arena.alloc("existing".to_owned());
207
208        let appended = arena.alloc_iter(
209            ["first", "duplicate", "duplicate"]
210                .into_iter()
211                .map(str::to_owned),
212        );
213
214        assert_eq!(existing.into_raw(), 0);
215        assert_eq!(arena[existing], "existing");
216        assert_eq!(appended.start.into_raw(), 1);
217        assert_eq!(appended.end.into_raw(), 4);
218        let appended_values: Vec<_> = Id::range_to_iter(appended.clone())
219            .map(|id| arena[id].as_str())
220            .collect();
221        assert_eq!(appended_values, ["first", "duplicate", "duplicate"]);
222
223        let empty = arena.alloc_iter(std::iter::empty::<String>());
224        assert_eq!(empty.start.into_raw(), 4);
225        assert_eq!(empty.end.into_raw(), 4);
226        assert_eq!(arena.data.len(), 4);
227        assert_eq!(arena[existing], "existing");
228    }
