15/// Current state of the PubGrub algorithm.
16#[derive(Clone)]
17pub(crate) struct State<DP: DependencyProvider> {
18    pub root_package: Id<DP::P>,
19    root_version: DP::V,
20
21    #[allow(clippy::type_complexity)]
22    incompatibilities: Map<Id<DP::P>, Vec<IncompDpId<DP>>>,
23
24    /// All incompatibilities expressing dependencies,
25    /// with common dependents merged.
26    #[allow(clippy::type_complexity)]
27    merged_dependencies: Map<(Id<DP::P>, Id<DP::P>), SmallVec<IncompDpId<DP>>>,
28
29    /// Partial solution.
30    /// TODO: remove pub.
31    pub(crate) partial_solution: PartialSolution<DP>,
32
33    /// The store is the reference storage for all incompatibilities.
34    pub(crate) incompatibility_store: Arena<Incompatibility<DP::P, DP::VS, DP::M>>,
35
36    /// The store is the reference storage for all packages.
37    pub(crate) package_store: HashArena<DP::P>,
38
39    /// This is a stack of work to be done in `unit_propagation`.
40    /// It can definitely be a local variable to that method, but
41    /// this way we can reuse the same allocation for better performance.
42    unit_propagation_buffer: SmallVec<Id<DP::P>>,
43}

45impl<DP: DependencyProvider> State<DP> {
46    /// Initialization of PubGrub state.
47    pub(crate) fn init(root_package: DP::P, root_version: DP::V) -> Self {
48        let mut incompatibility_store = Arena::new();
49        let mut package_store = HashArena::new();
50        let root_package = package_store.alloc(root_package);
51        let not_root_id = incompatibility_store.alloc(Incompatibility::not_root(
52            root_package,
53            root_version.clone(),
54        ));
55        let mut incompatibilities = Map::default();
56        incompatibilities.insert(root_package, vec![not_root_id]);
57        Self {
58            root_package,
59            root_version,
60            incompatibilities,
61            partial_solution: PartialSolution::empty(),
62            incompatibility_store,
63            package_store,
64            unit_propagation_buffer: SmallVec::Empty,
65            merged_dependencies: Map::default(),
66        }
67    }
68
69    /// Add the dependencies for the version being decided of the current package as
70    /// incompatibilities.
71    ///
72    /// `versions` is the set of versions that share these dependencies, in the simplest case the
73    /// version being decided: `VS::singleton(version)`. A caller that knows which versions exist
74    /// can widen the version being decided to a set that contains no other existing version,
75    /// e.g. with `Ranges::widen_versions`: If `2` is the only existing version between `1` and
76    /// `3`, the dependency incompatibilities for `{2}` can instead be created for `>1, <3`. When
77    /// resolution then rejects the version, the version sets derived from these
78    /// incompatibilities exclude a contiguous set instead of accumulating one hole per rejected
79    /// version, and the incompatibilities of adjacent versions merge into contiguous sets,
80    /// keeping the version sets and the operations on them minimal.
81    ///
82    /// `versions` must contain `version`, every existing version in `versions` must have exactly
83    /// the given dependencies, and the set of existing versions must not grow while resolution
84    /// is running.
85    pub(crate) fn add_package_version_dependencies(
86        &mut self,
87        package: Id<DP::P>,
88        version: DP::V,
89        versions: DP::VS,
90        dependencies: impl IntoIterator<Item = (DP::P, DP::VS)>,
91    ) -> Option<IncompId<DP::P, DP::VS, DP::M>> {
92        debug_assert!(
93            versions.contains(&version),
94            "the version being decided must be in the version set sharing its dependencies",
95        );
96        let dep_incompats =
97            self.add_incompatibility_from_dependencies(package, versions, dependencies);
98        self.partial_solution.add_package_version_incompatibilities(
99            package,
100            version,
101            dep_incompats,
102            &self.incompatibility_store,
103        )
104    }
105
106    /// Add an incompatibility to the state.
107    pub(crate) fn add_incompatibility(
108        &mut self,
109        mut incompat: Incompatibility<DP::P, DP::VS, DP::M>,
110    ) {
111        // Cached contradictions are only valid in the state that recorded them.
112        incompat.reset_contradiction_cache();
113        let id = self.incompatibility_store.alloc(incompat);
114        self.merge_incompatibility(id);
115    }

117    /// Add a single custom incompatibility that requires that the base package and the proxy
118    /// package share the same version range.
119    ///
120    /// This intended for cases where proxy packages (also known as virtual packages) are used.
121    /// Without this information, pubgrub does not know that these packages have to be at the same
122    /// version. In cases where the base package is already set to an incompatible version, this
123    /// avoids going through all versions of the proxy package. In cases where there are two
124    /// incompatible proxy packages, it avoids trying versions for both of them. This improves both
125    /// performance (we don't need to check all versions when there is a conflict) and error
126    /// messages (report a conflict of version ranges instead of enumerating the conflicting
127    /// versions).
128    ///
129    /// Using this method requires that each version of the proxy package depends on the exact
130    /// same version of the base package.
131    #[allow(unused)]
132    pub(crate) fn add_proxy_package_incompatibility(
133        &mut self,
134        proxy_package: Id<DP::P>,
135        base_package: Id<DP::P>,
136        versions: DP::VS,
137    ) {
138        let incompat = Incompatibility::from_dependency(
139            proxy_package,
140            versions.clone(),
141            (base_package, versions),
142        );
143        let id = self
144            .incompatibility_store
145            .alloc_iter([incompat].into_iter());
146        for id in IncompDpId::<DP>::range_to_iter(id) {
147            self.merge_incompatibility(id);
148        }
149    }

151    /// Add an incompatibility to the state.
152    #[cold]
153    pub(crate) fn add_incompatibility_from_dependencies(
154        &mut self,
155        package: Id<DP::P>,
156        versions: DP::VS,
157        deps: impl IntoIterator<Item = (DP::P, DP::VS)>,
158    ) -> std::ops::Range<IncompDpId<DP>> {
159        // Create incompatibilities and allocate them in the store.
160        let new_incompats_id_range =
161            self.incompatibility_store
162                .alloc_iter(deps.into_iter().map(|(dep_p, dep_vs)| {
163                    let dep_pid = self.package_store.alloc(dep_p);
164                    Incompatibility::from_dependency(package, versions.clone(), (dep_pid, dep_vs))
165                }));
166        // Merge the newly created incompatibilities with the older ones.
167        for id in IncompDpId::<DP>::range_to_iter(new_incompats_id_range.clone()) {
168            self.merge_incompatibility(id);
169        }
170        new_incompats_id_range
171    }

173    /// Unit propagation is the core mechanism of the solving algorithm.
174    /// CF <https://github.com/dart-lang/pub/blob/master/doc/solver.md#unit-propagation>
175    ///
176    /// For each package with a satisfied incompatibility, returns the package and the root cause
177    /// incompatibility.
178    #[cold]
179    #[allow(clippy::type_complexity)] // Type definitions don't support impl trait.
180    pub(crate) fn unit_propagation(
181        &mut self,
182        package: Id<DP::P>,
183    ) -> Result<SmallVec<(Id<DP::P>, IncompDpId<DP>)>, NoSolutionError<DP>> {
184        let mut satisfier_causes = SmallVec::default();
185        self.unit_propagation_buffer.clear();
186        self.unit_propagation_buffer.push(package);
187        while let Some(current_package) = self.unit_propagation_buffer.pop() {
188            // Iterate over incompatibilities in reverse order
189            // to evaluate first the newest incompatibilities.
190            let mut conflict_id = None;
191            // We only care about incompatibilities if it contains the current package.
192            for &incompat_id in self.incompatibilities[&current_package].iter().rev() {
193                if self
194                    .partial_solution
195                    .is_contradicted(&self.incompatibility_store[incompat_id])
196                {
197                    continue;
198                }
199                let current_incompat = &self.incompatibility_store[incompat_id];
200                match self.partial_solution.relation(current_incompat) {
201                    // If the partial solution satisfies the incompatibility
202                    // we must perform conflict resolution.
203                    Relation::Satisfied => {
204                        log::info!(
205                            "Start conflict resolution because incompat satisfied:\n   {}",
206                            current_incompat.display(&self.package_store)
207                        );
208                        conflict_id = Some(incompat_id);
209                        break;
210                    }
211                    Relation::AlmostSatisfied(package_almost) => {
212                        // Add `package_almost` to the `unit_propagation_buffer` set.
213                        // Putting items in `unit_propagation_buffer` more than once waste cycles,
214                        // but so does allocating a hash map and hashing each item.
215                        // In practice `unit_propagation_buffer` is small enough that we can just do a linear scan.
216                        if !self.unit_propagation_buffer.contains(&package_almost) {
217                            self.unit_propagation_buffer.push(package_almost);
218                        }
219                        // Add (not term) to the partial solution with incompat as cause.
220                        self.partial_solution.add_derivation(
221                            package_almost,
222                            incompat_id,
223                            &self.incompatibility_store,
224                        );
225                        // With the partial solution updated, the incompatibility is now contradicted.
226                        self.partial_solution
227                            .mark_contradicted(&mut self.incompatibility_store[incompat_id]);
228                    }
229                    Relation::Contradicted(_) => {
230                        self.partial_solution
231                            .mark_contradicted(&mut self.incompatibility_store[incompat_id]);
232                    }
233                    _ => {}
234                }
235            }
236            if let Some(incompat_id) = conflict_id {
237                let (package_almost, root_cause) = self
238                    .conflict_resolution(incompat_id, &mut satisfier_causes)
239                    .map_err(|terminal_incompat_id| {
240                        self.build_derivation_tree(terminal_incompat_id)
241                    })?;
242                self.unit_propagation_buffer.clear();
243                self.unit_propagation_buffer.push(package_almost);
244                // Add to the partial solution with incompat as cause.
245                self.partial_solution.add_derivation(
246                    package_almost,
247                    root_cause,
248                    &self.incompatibility_store,
249                );
250                // After conflict resolution and the partial solution update,
251                // the root cause incompatibility is now contradicted.
252                self.partial_solution
253                    .mark_contradicted(&mut self.incompatibility_store[root_cause]);
254            }
255        }
256        // If there are no more changed packages, unit propagation is done.
257        Ok(satisfier_causes)
258    }

260    /// Return the root cause or the terminal incompatibility. CF
261    /// <https://github.com/dart-lang/pub/blob/master/doc/solver.md#unit-propagation>
262    ///
263    /// When we found a conflict, we want to learn as much as possible from it, to avoid making (or
264    /// keeping) decisions that will be rejected. Say we found that the dependency requirements on X and the
265    /// dependency requirements on Y are incompatible. We may find that the decisions on earlier packages B and C
266    /// require us to make incompatible requirements on X and Y, so we backtrack until either B or C
267    /// can be revisited. To make it practical, we really only need one of the terms to be a
268    /// decision. We may as well leave the other terms general. Something like "the dependency on
269    /// the package X is incompatible with the decision on C" tends to work out pretty well. Then if
270    /// A turns out to also have a dependency on X the resulting root cause is still useful.
271    /// (`unit_propagation` will ensure we don't try that version of C.)
272    /// Of course, this is more heuristics than science. If the output is too general, then
273    /// `unit_propagation` will handle the confusion by calling us again with the next most specific
274    /// conflict it comes across. If the output is too specific, then the outer `solver` loop will
275    /// eventually end up calling us again until all possibilities are enumerated.
276    ///
277    /// To end up with a more useful incompatibility, this function combines incompatibilities into
278    /// derivations. Fulfilling this derivation implies the later conflict. By banning it, we
279    /// prevent the intermediate steps from occurring again, at least in the exact same way.
280    /// However, the statistics collected for `prioritize` may want to analyze those intermediate
281    /// steps. For example we might start with "there is no version 1 of Z", and
282    /// `conflict_resolution` may be able to determine that "that was inevitable when we picked
283    /// version 1 of X" which was inevitable when we picked W and so on, until version 1 of B, which
284    /// was depended on by version 1 of A. Therefore the root cause may simplify all the way down to
285    /// "we cannot pick version 1 of A". This will prevent us going down this path again. However
286    /// when we start looking at version 2 of A, and discover that it depends on version 2 of B, we
287    /// will want to prioritize the chain of intermediate steps to check if it has a problem with
288    /// the same shape. The `satisfier_causes` argument keeps track of these intermediate steps so
289    /// that the caller can use them for prioritization.
290    #[allow(clippy::type_complexity)]
291    #[cold]
292    fn conflict_resolution(
293        &mut self,
294        incompatibility: IncompDpId<DP>,
295        satisfier_causes: &mut SmallVec<(Id<DP::P>, IncompDpId<DP>)>,
296    ) -> Result<(Id<DP::P>, IncompDpId<DP>), IncompDpId<DP>> {
297        let mut current_incompat_id = incompatibility;
298        let mut current_incompat_changed = false;
299        loop {
300            if self.incompatibility_store[current_incompat_id]
301                .is_terminal(self.root_package, &self.root_version)
302            {
303                return Err(current_incompat_id);
304            } else {
305                let (package, satisfier_search_result) = self.partial_solution.satisfier_search(
306                    &self.incompatibility_store[current_incompat_id],
307                    &self.incompatibility_store,
308                );
309                match satisfier_search_result {
310                    SatisfierSearch::DifferentDecisionLevels {
311                        previous_satisfier_level,
312                    } => {
313                        self.backtrack(
314                            current_incompat_id,
315                            current_incompat_changed,
316                            previous_satisfier_level,
317                        );
318                        log::info!("backtrack to {previous_satisfier_level:?}");
319                        satisfier_causes.push((package, current_incompat_id));
320                        return Ok((package, current_incompat_id));
321                    }
322                    SatisfierSearch::SameDecisionLevels { satisfier_cause } => {
323                        let prior_cause = Incompatibility::prior_cause(
324                            current_incompat_id,
325                            satisfier_cause,
326                            package,
327                            &self.incompatibility_store,
328                        );
329                        log::info!("prior cause: {}", prior_cause.display(&self.package_store));
330                        current_incompat_id = self.incompatibility_store.alloc(prior_cause);
331                        satisfier_causes.push((package, current_incompat_id));
332                        current_incompat_changed = true;
333                    }
334                }
335            }
336        }
337    }

339    /// Backtracking.
340    fn backtrack(
341        &mut self,
342        incompat: IncompDpId<DP>,
343        incompat_changed: bool,
344        decision_level: DecisionLevel,
345    ) {
346        self.partial_solution.backtrack(decision_level);
347        if incompat_changed {
348            self.merge_incompatibility(incompat);
349        }
350    }
351
352    /// Add this incompatibility into the set of all incompatibilities.
353    ///
354    /// PubGrub collapses identical dependencies from adjacent package versions
355    /// into individual incompatibilities.
356    /// This substantially reduces the total number of incompatibilities
357    /// and makes it much easier for PubGrub to reason about multiple versions of packages at once.
358    ///
359    /// For example, rather than representing
360    /// foo 1.0.0 depends on bar ^1.0.0 and
361    /// foo 1.1.0 depends on bar ^1.0.0
362    /// as two separate incompatibilities,
363    /// they are collapsed together into the single incompatibility {foo ^1.0.0, not bar ^1.0.0}
364    /// (provided that no other version of foo exists between 1.0.0 and 2.0.0).
365    /// We could collapse them into { foo (1.0.0 ∪ 1.1.0), not bar ^1.0.0 }
366    /// without having to check the existence of other versions though.
367    fn merge_incompatibility(&mut self, mut id: IncompDpId<DP>) {
368        if let Some((p1, p2)) = self.incompatibility_store[id].as_dependency() {
369            // Self-dependencies cannot be merged.
370            if p1 != p2 {
371                let deps_lookup = self.merged_dependencies.entry((p1, p2)).or_default();
372                if let Some((past, merged)) =
373                    deps_lookup.as_mut_slice().iter_mut().find_map(|past| {
374                        self.incompatibility_store[id]
375                            .merge_dependents(&self.incompatibility_store[*past])
376                            .map(|merged| (past, merged))
377                    })
378                {
379                    let new = self.incompatibility_store.alloc(merged);
380                    for (pkg, _) in self.incompatibility_store[new].iter() {
381                        self.incompatibilities
382                            .entry(pkg)
383                            .or_default()
384                            .retain(|id| id != past);
385                    }
386                    *past = new;
387                    id = new;
388                } else {
389                    deps_lookup.push(id);
390                }
391            }
392        }
393        for (pkg, term) in self.incompatibility_store[id].iter() {
394            if cfg!(debug_assertions) {
395                assert_ne!(term, &crate::term::Term::any());
396            }
397            self.incompatibilities.entry(pkg).or_default().push(id);
398        }
399    }

403    fn build_derivation_tree(
404        &self,
405        incompat: IncompDpId<DP>,
406    ) -> DerivationTree<DP::P, DP::VS, DP::M> {
407        let mut all_ids: Set<IncompDpId<DP>> = Set::default();
408        let mut shared_ids = Set::default();
409        let mut stack = vec![incompat];
410        while let Some(i) = stack.pop() {
411            if let Some((id1, id2)) = self.incompatibility_store[i].causes() {
412                if all_ids.contains(&i) {
413                    shared_ids.insert(i);
414                } else {
415                    stack.push(id1);
416                    stack.push(id2);
417                }
418            }
419            all_ids.insert(i);
420        }
421        // To avoid recursion we need to generate trees in topological order.
422        // That is to say we need to ensure that the causes are processed before the incompatibility they effect.
423        // It happens to be that sorting by their ID maintains this property.
424        let mut sorted_ids = all_ids.into_iter().collect::<Vec<_>>();
425        sorted_ids.sort_unstable_by_key(|id| id.into_raw());
426        let mut precomputed = Map::default();
427        for id in sorted_ids {
428            let tree = Incompatibility::build_derivation_tree(
429                id,
430                &shared_ids,
431                &self.incompatibility_store,
432                &self.package_store,
433                &precomputed,
434            );
435            precomputed.insert(id, Arc::new(tree));
436        }
437        // Now the user can refer to the entire tree from its root.
438        Arc::into_inner(precomputed.remove(&incompat).unwrap()).unwrap()
439    }

459    /// A `resolve` loop that keeps the version sets minimal by widening the version whose
460    /// dependencies are added over the known versions.
461    ///
462    /// Returns `None` if there is no solution.
463    fn resolve_with_widening(
464        provider: &Provider,
465        root: &'static str,
466        version: u32,
467        widen: bool,
468    ) -> Option<BTreeMap<&'static str, u32>> {
469        // Sorted version lists per package, the input for the widening.
470        let versions: Map<&'static str, Vec<u32>> = provider
471            .packages()
472            .map(|p| (*p, provider.versions(p).unwrap().copied().collect()))
473            .collect();
474        let mut widened = false;
475
476        let mut state: State<Provider> = State::init(root, version);
477        let mut added_dependencies: Map<Id<&'static str>, BTreeSet<u32>> = Map::default();
478        let mut next = state.root_package;
479        let solution = loop {
480            state.unit_propagation(next).ok()?;
481
482            let Some((package, term_intersection)) =
483                state.partial_solution.pick_highest_priority_pkg(|p, r| {
484                    provider.prioritize(
485                        &state.package_store[p],
486                        r,
487                        &PackageResolutionStatistics::default(),
488                    )
489                })
490            else {
491                break state
492                    .partial_solution
493                    .extract_solution()
494                    .map(|(p, v)| (state.package_store[p], v))
495                    .collect();
496            };
497            next = package;
498
499            let Some(decision) = provider
500                .choose_version(&state.package_store[package], term_intersection)
501                .unwrap()
502            else {
503                let inc = Incompatibility::no_versions(
504                    package,
505                    Term::Positive(term_intersection.clone()),
506                );
507                state.add_incompatibility(inc);
508                continue;
509            };
510
511            if added_dependencies
512                .entry(package)
513                .or_default()
514                .insert(decision)
515            {
516                let dependencies = match provider
517                    .get_dependencies(&state.package_store[package], &decision)
518                    .unwrap()
519                {
520                    Dependencies::Unavailable(reason) => {
521                        state.add_incompatibility(Incompatibility::custom_version(
522                            package, decision, reason,
523                        ));
524                        continue;
525                    }
526                    Dependencies::Available(dependencies) => dependencies,
527                };
528                let version_set = NumVS::singleton(decision);
529                let versions = if widen {
530                    let widened_versions = version_set
531                        .widen_versions(versions.get(state.package_store[package]).unwrap());
532                    widened |= widened_versions != version_set;
533                    widened_versions
534                } else {
535                    version_set
536                };
537                state.add_package_version_dependencies(package, decision, versions, dependencies);
538            } else {
539                state.partial_solution.add_decision(package, decision);
540            }
541        };
542        if widen {
543            assert!(widened, "no version set was ever widened");
544        }
545        Some(solution)
546    }

548    /// A registry that requires rejecting versions of "a" and "b" one by one, similar to boto3 and
549    /// botocore: Without widening, the version sets of "a" accumulate one hole per rejected version,
550    /// and the incompatibilities of "a" versions depending on "j" merge into growing unions of
551    /// singletons.
552    fn backtracking_provider(conflict_below: u32) -> Provider {
553        let mut provider = Provider::new();
554        provider.add_dependencies(
555            "root",
556            1u32,
557            [
558                ("u", Ranges::strictly_lower_than(2u32)),
559                ("a", Ranges::full()),
560            ],
561        );
562        provider.add_dependencies("u", 1u32, []);
563        provider.add_dependencies("u", 2u32, []);
564        for k in 1..=30u32 {
565            provider.add_dependencies("a", k, [("b", Ranges::singleton(k)), ("j", Ranges::full())]);
566            // Versions of "b" above the threshold conflict with the "u" requirement of root.
567            let u_range = if k >= conflict_below {
568                Ranges::higher_than(2u32)
569            } else {
570                Ranges::strictly_lower_than(2u32)
571            };
572            provider.add_dependencies("b", k, [("u", u_range)]);
573        }
574        provider.add_dependencies("j", 1u32, []);
575        provider
576    }

578    #[test]
579    fn widening_finds_same_solution_after_backtracking() {
580        let provider = backtracking_provider(4);
581        let expected = BTreeMap::from([("root", 1), ("a", 3), ("b", 3), ("j", 1), ("u", 1)]);
582
583        let without = resolve_with_widening(&provider, "root", 1, false).unwrap();
584        let with = resolve_with_widening(&provider, "root", 1, true).unwrap();
585
586        assert_eq!(without, expected);
587        assert_eq!(with, expected);
588    }
